class _CounterScreenState extends State<CounterScreen> {

dart • Full Teacher's Provider Architecture 
import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

// ============================================================================
// 🔴 [PART 1: THE BUSINESS LOGIC CLASS (ChangeNotifier)]
// ============================================================================
class CounterProvider extends ChangeNotifier {
  int _count = 0;
  int get count => _count;

  void increment() {
    _count++;
    notifyListeners(); // ⭐ Tells listening Consumer widgets to redraw!
  }

  void decrement() {
    if (_count > 0) {
      _count--;
      notifyListeners();
    }
  }

  void reset() {
    _count = 0;
    notifyListeners();
  }
}

// ============================================================================
// 🟢 [PART 2: APP ROOT INJECTION]
// ============================================================================
void main() {
  runApp(
    ChangeNotifierProvider(
      create: (context) => CounterProvider(),
      child: const MaterialApp(
        debugShowCheckedModeBanner: false,
        home: ProviderCounterScreen(),
      ),
    ),
  );
}

5. Topic 3: SharedPreferences User Session

Key-Value Local Data Persistence
Topic 3
📝 Exam Question
"Create a User Login Session screen in Flutter that persists username and login state using SharedPreferences. Implement save, read on app start, and logout."
dart • Full Complete SharedPreferences Code 
import 'package:flutter/material.dart';
import 'package:shared_preferences/shared_preferences.dart';

void main() => runApp(const MaterialApp(
  debugShowCheckedModeBanner: false,
  home: UserSessionScreen(),
));

class UserSessionScreen extends StatefulWidget {
  const UserSessionScreen({super.key});

  @override
  State<UserSessionScreen> createState() => _UserSessionScreenState();
}

class _UserSessionScreenState extends State<UserSessionScreen> {
  // 🔴 [40% LOGIC: VARIABLES]
  final TextEditingController _usernameController = TextEditingController();
  String _savedUser = 'No user logged in';
  bool _isLoggedIn = false;

  @override
  void initState() {
    super.initState();
    _loadUserSession(); // ⭐ Read saved data on screen start
  }

  // 1. READ / LOAD DATA
  Future<void> _loadUserSession() async {
    final prefs = await SharedPreferences.getInstance();
    setState(() {
      _savedUser = prefs.getString('saved_user') ?? 'No user logged in';
      _isLoggedIn = prefs.getBool('is_logged_in') ?? false;
    });
  }

  // 2. WRITE / SAVE DATA
  Future<void> _saveUserSession() async {
    final username = _usernameController.text.trim();
    if (username.isEmpty) return;

    final prefs = await SharedPreferences.getInstance();
    await prefs.setString('saved_user', username);
    await prefs.setBool('is_logged_in', true);

    _usernameController.clear();
    _loadUserSession();
  }

  // 3. REMOVE / CLEAR DATA
  Future<void> _logout() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.remove('saved_user');
    await prefs.remove('is_logged_in');
    _loadUserSession();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('SharedPreferences Session'), centerTitle: true),
      body: Padding(
        padding: const EdgeInsets.all(20.0),
        child: Column(
          children: [
            TextField(
              controller: _usernameController,
              decoration: const InputDecoration(labelText: 'Enter Username', border: OutlineInputBorder()),
            ),
            const SizedBox(height: 16),
            Row(
              children: [
                Expanded(
                  child: ElevatedButton(onPressed: _saveUserSession, child: const Text('Save / Login')),
                ),
                const SizedBox(width: 12),
                Expanded(
                  child: ElevatedButton(
                    onPressed: _logout,
                    style: ElevatedButton.styleFrom(backgroundColor: Colors.red.shade100),
                    child: const Text('Logout / Clear'),
                  ),
                ),
              ],
            ),
            const SizedBox(height: 24),
            Card(
              elevation: 3,
              child: Padding(
                padding: const EdgeInsets.all(16.0),
                child: Column(
                  children: [
                    Text('Session: ${_isLoggedIn ? "LOGGED IN" : "LOGGED OUT"}',
                        style: TextStyle(fontWeight: FontWeight.bold, color: _isLoggedIn ? Colors.green : Colors.red)),
                    const SizedBox(height: 8),
                    Text('Saved User: $_savedUser', style: const TextStyle(fontSize: 18)),
                  ],
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }
}

6. Topic 4: SQFLite Relational CRUD Helper + Screen

SQLite Singleton • Table Creation • CRUD Operations
Topic 4
dart • DatabaseHelper Singleton + Screen 
import 'package:flutter/material.dart';
import 'package:sqflite/sqflite.dart';
import 'package:path/path.dart';

// ============================================================================
// 🔴 [PART 1: DATABASE HELPER SINGLETON - PURE DART CRUD CLASS]
// ============================================================================
class DatabaseHelper {
  static final DatabaseHelper instance = DatabaseHelper._init();
  static Database? _database;
  DatabaseHelper._init();

  Future<Database> get database async {
    if (_database != null) return _database!;
    _database = await _initDB('students.db');
    return _database!;
  }

  Future<Database> _initDB(String filePath) async {
    final dbPath = await getDatabasesPath();
    final path = join(dbPath, filePath);
    return await openDatabase(path, version: 1, onCreate: _createDB);
  }

  Future _createDB(Database db, int version) async {
    await db.execute('''
      CREATE TABLE students (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        rollNo TEXT NOT NULL
      )
    ''');
  }

  // [C] CREATE
  Future<int> insertStudent(Map<String, dynamic> row) async {
    final db = await database;
    return await db.insert('students', row);
  }

  // [R] READ
  Future<List<Map<String, dynamic>>> queryAllStudents() async {
    final db = await database;
    return await db.query('students');
  }

  // [U] UPDATE
  Future<int> updateStudent(Map<String, dynamic> row) async {
    final db = await database;
    int id = row['id'];
    return await db.update('students', row, where: 'id = ?', whereArgs: [id]);
  }

  // [D] DELETE
  Future<int> deleteStudent(int id) async {
    final db = await database;
    return await db.delete('students', where: 'id = ?', whereArgs: [id]);
  }
}

// ============================================================================
// 🟢 [PART 2: FLUTTER UI CONSUMING THE HELPER]
// ============================================================================
void main() => runApp(const MaterialApp(home: SqliteStudentScreen()));

class SqliteStudentScreen extends StatefulWidget {
  const SqliteStudentScreen({super.key});
  @override
  State<SqliteStudentScreen> createState() => _SqliteStudentScreenState();
}

class _SqliteStudentScreenState extends State<SqliteStudentScreen> {
  final _nameCtrl = TextEditingController();
  final _rollCtrl = TextEditingController();
  List<Map<String, dynamic>> _list = [];

  @override
  void initState() {
    super.initState();
    _refresh();
  }

  Future<void> _refresh() async {
    final data = await DatabaseHelper.instance.queryAllStudents();
    setState(() => _list = data);
  }

  Future<void> _add() async {
    if (_nameCtrl.text.isEmpty || _rollCtrl.text.isEmpty) return;
    await DatabaseHelper.instance.insertStudent({
      'name': _nameCtrl.text.trim(),
      'rollNo': _rollCtrl.text.trim(),
    });
    _nameCtrl.clear();
    _rollCtrl.clear();
    _refresh();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('SQFLite Student CRUD')),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          children: [
            TextField(controller: _nameCtrl, decoration: const InputDecoration(labelText: 'Name')),
            TextField(controller: _rollCtrl, decoration: const InputDecoration(labelText: 'Roll No')),
            const SizedBox(height: 10),
            ElevatedButton(onPressed: _add, child: const Text('Add Student')),
            const Divider(height: 30),
            Expanded(
              child: _list.isEmpty
                  ? const Center(child: Text('No records found'))
                  : ListView.builder(
                      itemCount: _list.length,
                      itemBuilder: (context, i) => ListTile(
                        title: Text(_list[i]['name']),
                        subtitle: Text('Roll: ${_list[i]["rollNo"]}'),
                        trailing: IconButton(
                          icon: const Icon(Icons.delete, color: Colors.red),
                          onPressed: () async {
                            await DatabaseHelper.instance.deleteStudent(_list[i]['id']);
                            _refresh();
                          },
                        ),
                      ),
                    ),
            ),
          ],
        ),
      ),
    );
  }
}

8. Topic 6: Form Validation with `GlobalKey<FormState>`

import 'package:flutter/material.dart';

void main() => runApp(const MaterialApp(home: FormScreen()));

class FormScreen extends StatefulWidget {
  const FormScreen({super.key});
  @override
  State<FormScreen> createState() => _FormScreenState();
}

class _FormScreenState extends State<FormScreen> {
  final _formKey = GlobalKey<FormState>();
  final _nameCtrl = TextEditingController();
  final _emailCtrl = TextEditingController();

  void _submit() {
    if (_formKey.currentState!.validate()) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Valid! Welcome ${_nameCtrl.text}'), backgroundColor: Colors.green),
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Form Validation')),
      body: Padding(
        padding: const EdgeInsets.all(20.0),
        child: Form(
          key: _formKey,
          child: Column(
            children: [
              TextFormField(
                controller: _nameCtrl,
                decoration: const InputDecoration(labelText: 'Name', border: OutlineInputBorder()),
                validator: (val) => (val == null || val.trim().isEmpty) ? 'Name is required' : null,
              ),
              const SizedBox(height: 16),
              TextFormField(
                controller: _emailCtrl,
                decoration: const InputDecoration(labelText: 'Email', border: OutlineInputBorder()),
                validator: (val) => (val == null || !val.contains('@')) ? 'Enter a valid email' : null,
              ),
              const SizedBox(height: 24),
              ElevatedButton(onPressed: _submit, child: const Text('Submit Form')),
            ],
          ),
        ),
      ),
    );
  }
}
